+# Newest mtime under a set of paths, as an epoch integer (0 when nothing matches).
+newest_mtime() {
+    local newest=0 f t
+    for f in "$@"; do
+        [ -e "$f" ] || continue
+        t=$(find "$f" -type f -printf '%T@\n' 2>/dev/null | cut -d. -f1 | sort -rn | head -1)
+        [ -n "$t" ] && [ "$t" -gt "$newest" ] && newest="$t"
+    done
+    echo "$newest"
+}

+newest_src_mtime() {

+}
